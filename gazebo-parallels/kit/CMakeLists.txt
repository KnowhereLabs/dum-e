cmake_minimum_required(VERSION 3.16)
project(gazebo_ogre_virgl)
find_package(gz-rendering8 8.2.3 EXACT REQUIRED)
find_package(gz-common5 REQUIRED COMPONENTS graphics)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OGRE REQUIRED IMPORTED_TARGET OGRE)
find_package(OpenGL REQUIRED)
set(RENDERING_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/src" CACHE PATH "Gazebo rendering source")
file(GLOB sources "${RENDERING_SOURCE_DIR}/ogre/src/*.cc")
add_library(gazebo_ogre_virgl SHARED ${sources})
target_compile_features(gazebo_ogre_virgl PRIVATE cxx_std_17)
target_include_directories(gazebo_ogre_virgl PRIVATE /usr/include/OGRE/Paging /usr/include/OGRE/Terrain /usr/include/OGRE/RTShaderSystem /usr/include/OGRE/Overlay)
target_compile_definitions(gazebo_ogre_virgl PRIVATE OGRE_VERSION_LT_1_11_0 OGRE_VERSION_LT_1_12_0 OGRE_RESOURCE_PATH="/usr/lib/aarch64-linux-gnu/OGRE-1.9.0")
target_link_libraries(gazebo_ogre_virgl PRIVATE gz-rendering8::gz-rendering8 gz-common5::requested gz-plugin2::register PkgConfig::OGRE OpenGL::GL OgreRTShaderSystem OgreOverlay OgreTerrain OgrePaging)
set_target_properties(gazebo_ogre_virgl PROPERTIES OUTPUT_NAME gz-rendering8-ogre)
add_custom_command(TARGET gazebo_ogre_virgl POST_BUILD COMMAND ${CMAKE_COMMAND} -E create_symlink libgz-rendering8-ogre.so $<TARGET_FILE_DIR:gazebo_ogre_virgl>/libgz-rendering-ogre.so)
target_include_directories(gazebo_ogre_virgl PRIVATE /usr/include/OGRE/RenderSystems/GL)

find_package(Qt5 REQUIRED COMPONENTS Core)
add_library(gazebo_virgl_defaults SHARED sources/defaults.cc)
target_link_libraries(gazebo_virgl_defaults PRIVATE dl Qt5::Core)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_executable(refresh_gui_state tools/refresh_gui_state.cc)
target_link_libraries(refresh_gui_state PRIVATE gz-transport13::core gz-msgs10::gz-msgs10)
